// 科目颜色：与「成绩折线图.html」保持一致的配色规则——按名称识别，常见科目用固定配色，
// 其余任意科目名称按出现顺序从备用调色板中取色，保证同名科目全程颜色一致。
var CURATED_COLORS = {
  '语文':'#e8a33d', '数学':'#d9645f', '英语':'#7c8fe0',
  '物理':'#52b08a', '化学':'#a97cd1', '地理':'#d4b483',
  '生物':'#4fb8a3', '政治':'#e0789a', '历史':'#c9a24d'
};
var FALLBACK_PALETTE = ['#6f9cde','#e0955a','#8fbf6f','#c98ad9','#5fb3c9','#d97a7a','#9aa84f','#b98fd9','#5fa8d9','#d9b25f'];
function colorForSubjectName(name, allNamesInOrder){
  if(CURATED_COLORS[name]) return CURATED_COLORS[name];
  var customNames = allNamesInOrder.filter(function(n){ return !CURATED_COLORS[n]; });
  var idx = customNames.indexOf(name);
  return FALLBACK_PALETTE[(idx<0?0:idx) % FALLBACK_PALETTE.length];
}
function allSubjectNamesInOrder(){
  var seen = {}, list = [];
  state.exams.forEach(function(ex){
    ex.subjects.forEach(function(s){
      var n = (s.name||'').trim();
      if(n && !seen[n]){ seen[n]=true; list.push(n); }
    });
  });
  return list;
}

/* ---------------- import ---------------- */
function extractExamData(htmlText){
  var re = /<script id="exam-data" type="application\/json">([\s\S]*?)<\/script>/;
  var m = htmlText.match(re);
  if(!m) return null;
  try{
    return JSON.parse(m[1]);
  }catch(e){
    console.error(e);
    return null;
  }
}

function handleFile(file){
  if(!file){ return; }
  if(!/\.html?$/i.test(file.name)){
    showToast('请选择 .html 格式的成绩折线图文件', true);
    return;
  }
  var reader = new FileReader();
  reader.onload = function(e){
    var text = e.target.result;
    var data = extractExamData(text);
    if(data === null){
      setStatus('未在文件中找到成绩数据，请确认这是「成绩折线图.html」', 'err');
      showToast('导入失败：未找到 exam-data 数据段', true);
      return;
    }
    state.originalHtml = text;
    state.exams = normalizeExams(data);
    setStatus('已成功导入：' + file.name + '（共 ' + state.exams.length + ' 场考试）', 'ok');
    showToast('导入成功，共读取到 ' + state.exams.length + ' 场考试');
    document.getElementById('btnExport').disabled = false;
    state.openIndex = state.exams.length ? state.exams.length - 1 : null;
    renderExamList();
  };
  reader.onerror = function(){ showToast('文件读取失败', true); };
  reader.readAsText(file, 'utf-8');
}

// 科目以“名称”为唯一标识（不再依赖内部编号），无论是导入旧版文件（曾用 key 字段）
// 还是填写任意自定义科目名，都能被正确识别与连续追踪。
function normalizeExams(data){
  if(!Array.isArray(data)) return [];
  return data.map(function(ex){
    return {
      id: ex.id || uid(),
      name: ex.name || '未命名考试',
      date: ex.date || '',
      classRank: ex.classRank === undefined ? null : ex.classRank,
      gradeRank: ex.gradeRank === undefined ? null : ex.gradeRank,
      classSize: ex.classSize === undefined ? null : ex.classSize,
      gradeSize: ex.gradeSize === undefined ? null : ex.gradeSize,
      subjects: Array.isArray(ex.subjects) ? ex.subjects.map(function(s){
        return {
          _uid: subUid(),
          name: (s.name || '科目').trim(),
          full: s.full === undefined ? 100 : s.full,
          raw: s.raw === undefined ? 0 : s.raw,
          converted: (s.converted === undefined) ? null : s.converted,
          useConverted: !!s.useConverted
        };
      }) : []
    };
  }).sort(function(a,b){ return new Date(a.date||0) - new Date(b.date||0); });
}

    copy.subjects.forEach(function(s){ s._uid = subUid(); });

function makeInput(type, value, onChange, placeholder){
  var input = document.createElement('input');
  input.type = type;
  if(value !== null && value !== undefined) input.value = value;
  if(placeholder) input.placeholder = placeholder;
  if(type === 'number') input.step = 'any';
  input.addEventListener('input', function(){
    onChange(input.value);
  });
  return input;
}

      state.openIndex = i;
    });
    quickRow.appendChild(customBtn);
  }
  refreshQuickAddRow();
  wrap.appendChild(quickRow);

  var previewWrap = document.createElement('div');
  previewWrap.id = 'total-preview-' + i;
  previewWrap.className = 'total-preview';
  wrap.appendChild(previewWrap);
  updateTotalPreview(previewWrap, exam);

  return wrap;
}

function updateTotalPreview(el, exam){
  var total = examTotal(exam);
  var full = exam.subjects.reduce(function(s,x){return s+(Number(x.full)||0);},0);
  el.innerHTML =
    '<span>本次总分：<b>' + (Math.round(total*10)/10) + '</b> / ' + full + '</span>' +
    '<span>科目数：<b>' + exam.subjects.length + '</b></span>';
}

function syncTotal(i){
  var el = document.getElementById('total-preview-' + i);
  if(el) updateTotalPreview(el, state.exams[i]);
  syncHeader(i);
}

function syncHeader(i){
  var cards = document.querySelectorAll('.exam-card');
  var card = cards[i];
  if(!card) return;
  var exam = state.exams[i];
  var metaEl = card.querySelector('.exam-card-meta');
  var titleEl = card.querySelector('.exam-card-title');
  if(titleEl){
    titleEl.childNodes[1].textContent = ' ' + (exam.name || '未命名考试');
  }
  if(metaEl){
    metaEl.textContent = (exam.date||'未设置日期') + ' · 总分 ' + Math.round(examTotal(exam)*10)/10;
  }
}

function escapeHtml(s){
  return String(s).replace(/[&<>"']/g, function(c){
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];
  });
}

/* ---------------- add exam ---------------- */
function addExam(){
  var last = state.exams[state.exams.length-1];
  var newExam = {
    id: uid(),
    name: '新考试',
    date: new Date().toISOString().slice(0,10),
    classRank: null,
    gradeRank: null,
    classSize: last ? last.classSize : null,
    gradeSize: last ? last.gradeSize : null,
    subjects: (last ? last.subjects : QUICK_SUBJECTS.slice(0,6)).map(function(s){
      return {_uid: subUid(), name:s.name, full:s.full, raw:0, converted:null, useConverted:false};
    })
  };
  state.exams.push(newExam);
  state.openIndex = state.exams.length - 1;
  if(!state.originalHtml){ state.originalHtml = BASE_TEMPLATE; document.getElementById('btnExport').disabled = false; setStatus('已新建空白成绩单', 'ok'); }
  renderExamList();
  var cards = document.querySelectorAll('.exam-card');
  var target = cards[cards.length-1];
  if(target) target.scrollIntoView({behavior: 'smooth', block:'center'});
  showToast('已新增考试，请填写科目成绩（若有新科目，可用"快捷添加"或直接改科目名称）');
}

/* ---------------- export ---------------- */
function exportHtml(){
  if(!state.originalHtml){ showToast('请先导入或新建成绩单', true); return; }
  var sortedExams = state.exams.slice().sort(function(a,b){ return new Date(a.date||0) - new Date(b.date||0); });
  var cleaned = sortedExams.map(function(ex){
    return {
      id: ex.id,
      name: ex.name,
      date: ex.date,
      classRank: ex.classRank,
      gradeRank: ex.gradeRank,
      classSize: ex.classSize,
      gradeSize: ex.gradeSize,
      subjects: ex.subjects.map(function(s){
        return {
          name: (s.name||'').trim(), full: Number(s.full)||0,
          raw: Number(s.raw)||0,
          converted: (s.converted===null||s.converted===undefined||s.converted==='') ? null : Number(s.converted),
          useConverted: !!s.useConverted
        };
      })
    };
  });
  var jsonStr = JSON.stringify(cleaned, null, 2);
  var re = /(<script id="exam-data" type="application\/json">)([\s\S]*?)(<\/script>)/;
  var output = state.originalHtml.replace(re, function(match, p1, p2, p3){
    return p1 + '\n' + jsonStr + '\n' + p3;
  });

  var blob = new Blob([output], {type:'text/html;charset=utf-8'});
  var url = URL.createObjectURL(blob);
  var a = document.createElement('a');
  a.href = url;
  a.download = '成绩折线图.html';
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  setTimeout(function(){ URL.revokeObjectURL(url); }, 4000);
  showToast('已导出「成绩折线图.html」，请查看下载文件夹');
}
